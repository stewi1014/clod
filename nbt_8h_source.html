220char *nbt_payload(char *tag, const char type, const char *end) {
221 __nbt_assert(nbt_type_has_payload(type));
222 if (!__nbt_has_data(tag, end, 1)) return nullptr;
223 if (tag == nullptr || tag[0] != type) return nullptr;
224
225 if (!__nbt_has_data(tag, end, 3)) return nullptr;
226 const uint16_t name_size =
227 ((uint16_t)(unsigned char)tag[1]<<8) +
228 ((uint16_t)(unsigned char)tag[2]);
229
230 if (!__nbt_has_data(tag, end, 3 + name_size)) return nullptr;
231 tag += 3 + name_size;
232
233 switch (type){
234 case NBT_BYTE: if (!__nbt_has_data(tag, end, 1)) return nullptr; break;
235 case NBT_SHORT: if (!__nbt_has_data(tag, end, 2)) return nullptr; break;
236 case NBT_INT: if (!__nbt_has_data(tag, end, 4)) return nullptr; break;
237 case NBT_LONG: if (!__nbt_has_data(tag, end, 8)) return nullptr; break;
238 case NBT_FLOAT: if (!__nbt_has_data(tag, end, 4)) return nullptr; break;
239 case NBT_DOUBLE: if (!__nbt_has_data(tag, end, 8)) return nullptr; break;
240 case NBT_BYTE_ARRAY: if (!__nbt_has_data(tag, end, 4)) return nullptr; break;
241 case NBT_STRING: if (!__nbt_has_data(tag, end, 2)) return nullptr; break;
242 case NBT_LIST: if (!__nbt_has_data(tag, end, 5)) return nullptr; break;
243 case NBT_INT_ARRAY: if (!__nbt_has_data(tag, end, 4)) return nullptr; break;
244 case NBT_LONG_ARRAY: if (!__nbt_has_data(tag, end, 4)) return nullptr; break;
245 }
246
247 return tag;
248}

464uint64_t nbt_packed_array_elem(char *payload, const int i, const int bits) {
465 __nbt_assert(payload != nullptr);
466
467 const auto size = nbt_long_array_size(payload);
468 payload = nbt_long_array_payload(payload);
469
470 const int c = 64 / bits;
471
472 __nbt_assert((i / c) < size);
473
474 payload += (i / c) * 8;
475 const uint64_t n =
476 ((uint64_t)(unsigned char)payload[0] << 56) +
477 ((uint64_t)(unsigned char)payload[1] << 48) +
478 ((uint64_t)(unsigned char)payload[2] << 40) +
479 ((uint64_t)(unsigned char)payload[3] << 32) +
480 ((uint64_t)(unsigned char)payload[4] << 24) +
481 ((uint64_t)(unsigned char)payload[5] << 16) +
482 ((uint64_t)(unsigned char)payload[6] << 8) +
483 ((uint64_t)(unsigned char)payload[7]);
484
485 return n >> i % c * bits & ((~0ULL) >> (64 - bits));
486}
